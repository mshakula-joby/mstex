\begin{frame}{Theorem blocks}
	\begin{theorem}[Cauchy--Schwarz]
		For all vectors $u, v$ in an inner product space, $|\langle u, v\rangle|^{2} \le \langle u, u\rangle\,\langle v, v\rangle$.
	\end{theorem}
	\begin{proof}
		Expand $\|u - tv\|^{2} \ge 0$ and choose $t = \langle u, v\rangle / \langle v, v\rangle$.
	\end{proof}
